async function sendNotification(teacherIds, notificationData) {
  for (const teacherId of teacherIds) {
    try {
      const teacherDoc = await db.collection("teachers").doc(teacherId).get();
      if (teacherDoc.exists && teacherDoc.data().fcm_token) {
        const message = {
          token: teacherDoc.data().fcm_token,
          notification: notificationData,
          data: { type: notificationData.type || "general", teacher_id: teacherId }
        };
        await admin.messaging().send(message);
        console.log("Notification sent to:", teacherId);
      }
    } catch (error) {
      console.error("Notification failed for:", teacherId, error);
    }
  }
}
